// REGRAS COMPLETAS DO LABORATÓRIO. Substituir o conteúdo inteiro no Firebase Console.
// Somente dados fictícios; não autoriza coleções de produção.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /rom_lab_conexao/{userId} {
      allow get: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['dono', 'tipo', 'valor', 'atualizadoEm'])
        && request.resource.data.keys().hasAll(['dono', 'tipo', 'valor', 'atualizadoEm'])
        && request.resource.data.dono == userId
        && request.resource.data.tipo == 'teste_conexao'
        && request.resource.data.valor == 1;
      allow list, delete: if false;
    }
    match /rom_lab_sync/{userId} {
      allow get: if request.auth != null && request.auth.uid == userId;
      allow create, update: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.keys().hasOnly(['dono', 'palete', 'observacao', 'origem', 'atualizadoEm'])
        && request.resource.data.keys().hasAll(['dono', 'palete', 'observacao', 'origem', 'atualizadoEm'])
        && request.resource.data.dono == userId
        && request.resource.data.palete == 'TESTE-SYNC-001'
        && request.resource.data.origem == 'laboratorio'
        && request.resource.data.observacao is string
        && request.resource.data.observacao.size() > 0
        && request.resource.data.observacao.size() <= 100;
      allow list, delete: if false;
    }
    match /rom_lab_paletes/{paleteId} {
      function permitido() {
        return request.auth != null && paleteId == 'TESTE-CONCORRENCIA-001';
      }
      function saldoValido(d) {
        return d.saldo is int && d.saldo >= 0 && d.saldo <= 50
          && d.inicial == 50 && d.codigo == paleteId;
      }
      allow get: if permitido();
      allow list, delete: if false;
      allow create: if permitido()
        && request.resource.data.keys().hasOnly(['codigo', 'saldo', 'inicial', 'criadoEm'])
        && request.resource.data.keys().hasAll(['codigo', 'saldo', 'inicial', 'criadoEm'])
        && request.resource.data.saldo == 50
        && saldoValido(request.resource.data);
      allow update: if permitido()
        && resource.data.codigo == paleteId
        && resource.data.inicial == 50
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['saldo', 'atualizadoEm'])
        && request.resource.data.keys().hasOnly(['codigo', 'saldo', 'inicial', 'criadoEm', 'atualizadoEm'])
        && saldoValido(request.resource.data)
        && request.resource.data.saldo < resource.data.saldo;
      match /movimentos/{operacaoId} {
        allow get, list: if request.auth != null && paleteId == 'TESTE-CONCORRENCIA-001';
        allow create: if request.auth != null
          && paleteId == 'TESTE-CONCORRENCIA-001'
          && operacaoId.size() >= 8 && operacaoId.size() <= 80
          && request.resource.data.keys().hasOnly(['operacao', 'quantidade', 'operador', 'criadoEm'])
          && request.resource.data.keys().hasAll(['operacao', 'quantidade', 'operador', 'criadoEm'])
          && request.resource.data.operacao == operacaoId
          && request.resource.data.operador == request.auth.uid
          && request.resource.data.quantidade is int
          && request.resource.data.quantidade >= 1
          && request.resource.data.quantidade <= 50;
        allow update, delete: if false;
      }
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
// IMPORTANTE: estas regras limitam os dados de teste, mas não comprovam que um
// cliente adulterado não consiga gerar divergência entre saldo e histórico.
// Segurança definitiva exige validação atômica no backend da solução escolhida.
