match /rom_lab_multiplos/{paleteId} {
  allow read: if request.auth != null && paleteId in ['LAB-A','LAB-B','LAB-C'];
  allow create: if request.auth != null
    && paleteId in ['LAB-A','LAB-B','LAB-C']
    && request.resource.data.keys().hasOnly(['codigo','sku','lote','destino','estoque','romaneio','historico'])
    && request.resource.data.codigo == paleteId
    && request.resource.data.estoque is int
    && request.resource.data.estoque >= 0
    && request.resource.data.romaneio == 0
    && request.resource.data.historico is list
    && request.resource.data.historico.size() == 1;
  allow update: if request.auth != null
    && paleteId in ['LAB-A','LAB-B','LAB-C']
    && request.resource.data.keys().hasOnly(['codigo','sku','lote','destino','estoque','romaneio','historico'])
    && request.resource.data.codigo == resource.data.codigo
    && request.resource.data.sku == resource.data.sku
    && request.resource.data.lote == resource.data.lote
    && request.resource.data.destino == resource.data.destino
    && request.resource.data.estoque is int
    && request.resource.data.romaneio is int
    && request.resource.data.estoque >= 0
    && request.resource.data.romaneio >= 0
    && request.resource.data.historico is list
    && request.resource.data.historico.size() == resource.data.historico.size() + 1
    && request.resource.data.historico.size() <= 40;
  allow delete: if false;
}
