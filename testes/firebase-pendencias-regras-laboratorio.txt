rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
  match /rom_lab_conexao/{userId} {
   allow get: if request.auth != null && request.auth.uid == userId;
   allow create, update: if request.auth != null && request.auth.uid == userId
     && request.resource.data.keys().hasOnly(['dono','tipo','valor','atualizadoEm'])
     && request.resource.data.keys().hasAll(['dono','tipo','valor','atualizadoEm'])
     && request.resource.data.dono == userId
     && request.resource.data.tipo == 'teste_conexao'
     && request.resource.data.valor == 1;
   allow list, delete: if false;
  }
  match /rom_lab_sync/{userId} {
   allow get: if request.auth != null && request.auth.uid == userId;
   allow create, update: if request.auth != null && request.auth.uid == userId
     && request.resource.data.keys().hasOnly(['dono','palete','observacao','origem','atualizadoEm'])
     && request.resource.data.keys().hasAll(['dono','palete','observacao','origem','atualizadoEm'])
     && request.resource.data.dono == userId
     && request.resource.data.palete == 'TESTE-SYNC-001'
     && request.resource.data.origem == 'laboratorio'
     && request.resource.data.observacao is string
     && request.resource.data.observacao.size() > 0
     && request.resource.data.observacao.size() <= 100;
   allow list, delete: if false;
  }
  match /rom_lab_pendencias/{cenario} {
   function autorizado(){return request.auth != null && cenario == 'CENARIO-001';}
   function dadosValidos(d){
    return d.keys().hasOnly(['codigo','estoque','romaneio','historico','atualizadoEm'])
      && d.keys().hasAll(['codigo','estoque','romaneio','historico','atualizadoEm'])
      && d.codigo == 'CENARIO-001'
      && d.estoque is int && d.estoque >= 0 && d.estoque <= 100000
      && d.romaneio is int && d.romaneio >= 0 && d.romaneio <= 100000
      && d.historico is list && d.historico.size() >= 1 && d.historico.size() <= 60;
   }
   allow get: if autorizado();
   allow list, delete: if false;
   allow create: if autorizado() && dadosValidos(request.resource.data)
     && request.resource.data.estoque == 30 && request.resource.data.romaneio == 0
     && request.resource.data.historico == ['Cenário inicializado: estoque 30, Romaneio 0'];
   allow update: if autorizado() && dadosValidos(request.resource.data)
     && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['estoque','romaneio','historico','atualizadoEm'])
     && request.resource.data.historico.size() == resource.data.historico.size()+1
     && request.resource.data.historico.hasAll(resource.data.historico);
  }
  match /{document=**} {allow read, write: if false;}
 }
}
