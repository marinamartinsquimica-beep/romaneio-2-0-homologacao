match /rom_lab_duplicidade14/{paleteId} {
 function autorizado() {
  return request.auth != null && paleteId in ['LAB-A','LAB-B'];
 }
 function valido(d) {
  return d.keys().hasOnly(['codigo','sku','lote','estoque','romaneio','operacoes','historico'])
   && d.keys().hasAll(['codigo','sku','lote','estoque','romaneio','operacoes','historico'])
   && d.codigo == paleteId && d.sku == 'G' && d.lote == '001'
   && d.estoque == 20 && d.romaneio is int
   && d.romaneio >= 0 && d.romaneio <= 100000
   && d.operacoes is list && d.operacoes.size() <= 25
   && d.historico is list && d.historico.size() <= 26;
 }
 allow get: if autorizado();
 allow list, delete: if false;
 allow create: if autorizado() && valido(request.resource.data)
  && request.resource.data.romaneio == 0
  && request.resource.data.operacoes == []
  && request.resource.data.historico == ['Palete inicializado'];
 allow update: if autorizado() && valido(request.resource.data)
  && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['romaneio','operacoes','historico'])
  && request.resource.data.operacoes.size() == resource.data.operacoes.size()+1
  && request.resource.data.historico.size() == resource.data.historico.size()+1
  && request.resource.data.operacoes.hasAll(resource.data.operacoes)
  && request.resource.data.historico.hasAll(resource.data.historico);
}
